# -*- coding: utf-8 -*-
# Veri tiplerini doğrulamak ve API giriş/çıkış şablonlarını oluşturmak için Pydantic kullanıyoruz
from pydantic import BaseModel
from datetime import datetime
from typing import Optional, List

# --- Kontrol Sonuçları Şemaları ---

# Kontrol sonuçlarının temel şeması (ortak alanlar)
class CheckResultBase(BaseModel):
    status_code: Optional[int] = None       # int gelebilir veya çökme durumunda None olabilir
    response_time_ms: Optional[float] = None # milisaniye süresi (sayı veya None)
    is_up: bool                             # True veya False olmak zorunda
    error_message: Optional[str] = None     # Varsa hata mesajı, yoksa None

# Kullanıcıya kontrol sonucunu dönerken kullandığımız şema (ID ve zaman eklenir)
class CheckResultResponse(CheckResultBase):
    id: int
    target_id: int
    timestamp: datetime

    class Config:
        from_attributes = True  # SQLAlchemy ORM nesnesini otomatik olarak JSON verisine çevirir


# --- Hedef Site Şemaları ---

# Hedef sitenin temel şeması (kullanıcıdan beklenen temel bilgiler)
class TargetBase(BaseModel):
    name: str  # Site ismi zorunlu metin
    url: str   # Site linki zorunlu metin

# Kullanıcı POST ile yeni site eklerken kullanılacak şema:
# Sadece name ve url bekler, id veya checks gönderilemez.
class TargetCreate(TargetBase):
    pass

# Kullanıcıya site bilgisini API üzerinden geri dönerken kullandığımız şema:
class TargetResponse(TargetBase):
    id: int                                # Veritabanının atadığı ID
    checks: List[CheckResultResponse] = [] # Bu siteye ait geçmiş ölçümlerin listesi

    class Config:
        from_attributes = True  # ORM modelini JSON formatına çevirmeyi sağlar
